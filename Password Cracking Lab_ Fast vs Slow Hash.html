สังเกตเวลา: MD5/SHA-1 คำนวณเสร็จใน <1 ms ส่วน bcrypt (cost factor ปรับได้) ใช้เวลาหลัก 10–300+ ms ต่อครั้ง — นี่คือหัวใจของ "Adaptive Hash"

สาธิตว่า Salt ทำให้รหัสผ่านเดียวกันได้ hash ต่างกันทุกครั้ง (ป้องกัน rainbow table) และ Key Stretching (เช่น bcrypt, PBKDF2) ทำให้การเดาแต่ละครั้งช้าลงโดยตั้งใจ

ตัวอย่างโค้ด Python สำหรับ Production: ใช้ bcrypt (salt ในตัว + adaptive cost)
import bcrypt

def hash_password(password: str, rounds: int = 12) -> bytes:
    salt = bcrypt.gensalt(rounds=rounds)      # สุ่ม salt ใหม่ทุกครั้ง
    return bcrypt.hashpw(password.encode(), salt)

def verify_password(password: str, hashed: bytes) -> bool:
    return bcrypt.checkpw(password.encode(), hashed)

stored = hash_password("MySecret!2024", rounds=12)
print(stored)                                  # เก็บค่านี้ลงฐานข้อมูล
print(verify_password("MySecret!2024", stored))  # True
print(verify_password("wrongpass", stored))       # False
### ทางเลือก: PBKDF2-HMAC (Key Stretching ด้วย hashlib มาตรฐาน)
import hashlib, os

def hash_pbkdf2(password: str, iterations: int = 310_000):
    salt = os.urandom(16)
    dk = hashlib.pbkdf2_hmac("sha256", password.encode(), salt, iterations)
    return salt, dk

salt, dk = hash_pbkdf2("MySecret!2024")
# ยิ่ง iterations สูง ยิ่งช้าสำหรับทั้งฝั่ง login (ครั้งเดียว) และฝั่งแฮกเกอร์ (ล้านครั้ง)
