Python Secure Hashing Implementation 

import bcrypt
import time

# 1. ฟังก์ชันสำหรับการลงทะเบียนรหัสผ่านอย่างปลอดภัย (Salting + Key Stretching)
def hash_secure_password(plain_text_password: str) -> bytes:
    # แปลงข้อความให้อยู่ในรูปแบบ Bytes
    password_bytes = plain_text_password.encode('utf-8')
    
    # กำหนด Cost Factor / Rounds = 12 (มาตรฐานความปลอดภัยปี 2024)
    # gensalt() จะทำการสร้างค่า Salt แบบสุ่มขนาด 128-bit โดยอัตโนมัติ
    salt = bcrypt.gensalt(rounds=12)
    
    # ทำการ Hash โดยใช้ Bcrypt Algorithm (หน่วงเวลาประมวลผล ~0.25 วินาที)
    hashed_password = bcrypt.hashpw(password_bytes, salt)
    return hashed_password

# 2. ฟังก์ชันสำหรับการ ตรวจสอบการเข้าสู่ระบบ (Login Verification)
def verify_password(plain_text_password: str, stored_hash: bytes) -> bool:
    password_bytes = plain_text_password.encode('utf-8')
    
    # checkpw จะดึง Salt และ Cost Factor จาก stored_hash มาคำนวณเปรียบเทียบ
    return bcrypt.checkpw(password_bytes, stored_hash)

# --- ทดสอบการทำงาน ---
user_pwd = "P@ssw0rd123"
db_hash = hash_secure_password(user_pwd)

print(f"ค่าที่บันทึกลง Database: {db_hash.decode('utf-8')}")
print(f"ผลการยืนยันรหัสผ่าน: {verify_password(user_pwd, db_hash)}")
คำอธิบายหลักการทำงาน: โค้ดนี้ใช้ไลบรารี bcrypt ซึ่งมีระบบสร้าง Salt สุ่มให้อัตโนมัติในตัวเพื่อป้องกัน Rainbow Table และใช้ Work Factor เพื่อชะลอการเจาะรหัสผ่านของแฮกเกอร์

มาตรฐาน OWASP Password Storage Guidelines

1. การใช้ Cryptographic Salt (Salting)

ต้องเพิ่ม Salt สุ่มความยาวอย่างน้อย 16 bytes ให้ผู้ใช้แต่ละคนแบบ Unique เพื่อทำลายการใช้ Rainbow Table

2. การใช้ Key Stretching / Work Factor

ใช้ฟังก์ชัน Adaptive Hash ที่วนลูปคำนวณ เช่น Bcrypt (Cost 12+), PBKDF2 (600,000+ rounds)

3. ใช้ Memory-Hard Functions (Argon2id)

มาตรฐานสูงสุดแนะนำ Argon2id เนื่องจากใช้หน่วยความจำ (RAM) สูง ป้องกันการแคร็กด้วยการ์ดจอ (GPU) และชิปเฉพาะทาง (ASIC)

4. การผสานความปลอดภัยหลายชั้น (Pepper & MFA)

เพิ่ม Pepper แยกเก็บไว้นอกฐานข้อมูล และบังคับใช้ Multi-Factor Authentication (2FA) สำหรับทุกบัญชีสำคัญ
